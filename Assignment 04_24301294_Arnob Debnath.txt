public class Shape2D{
  public double AR;//area
  public int AK;//length
  public String AD;//Area type
  public Shape2D(){//contructor without parameter
    this.AK=5;
    this.AD="Square";
    this.AR=AK*AK;
    System.out.println(" A"+AD+" has been created with length:"+ AK);
  }
  public Shape2D(int am, int an){//constructor with 2 parameters which is a rectangle
    this.AD=" Rectangle";
    this.AR=am*an;
    System.out.println(" A"+AD+" has been created with length:"+ am+"and breadth:"+an);
  }
  public Shape2D(int am, int an, String AF){//constructor with 3 parameters whics is either rectangle or triangle
    if(AF.equals("Triangle")){
      this.AD="Triangle";
      this.AR=0.5*am*an;
      System.out.println(" A"+AD+" has been created with height:"+ am+"and base:"+an);
  }
    else{
    this.AD=" Rectangle";
    this.AR=am*an;
    System.out.println(" A"+AD+" has been created with length:"+ am+"and breadth:"+an);
    }
  }
  public Shape2D(int am, int an, int mo){//contructor with 3 parameters which is a triangle
    this.AD="Triangle";
    double xm=(am+an+mo)/2;
    double mx=xm*(xm-am)*(xm-an)*(xm-mo);
    this.AR=Math.sqrt(mx);
    System.out.println("A Triangle has been created with the following sides:"+am+","+an+","+mo);
  }
  public void area(){
    System.out.println("The area of the" +AD+"is:"+AR);
  }
                         
}

public class Triangle{
  public int AR1;//side1
  public int AR2;//side2
  public int AR3;//side3
  public int AD;//perimeter
  public void updateSides(int ar1,int ar2,int ar3){//method for taking side measurement
    this.AR1=ar1;
    this.AR2=ar2;
    this.AR3=ar3;
    this.AD=AR1+AR2+AR3;
  }
  public void triangleDetails(){//method for showing side and peremeter
   System.out.println(" Three sides of the triangle are:"+AR1+","+AR2+","+AR3);
   System.out.println("Perimeter"+AD);
  }
  public String printTriangleType(){//method for showing triangle type
    if(AR1==AR2&&AR1==AR3){
      return "This is an Equilateral Triangle.";
    }
    else if(((AR1==AR2)&&(AR1!=AR3))||((AR2==AR3)&&(AR2!=AR1))||((AR3==AR1)&&(AR3!=AR2))){
      return "This is an Isosceles Triangle.";
    }
    else{
      return"This is an Scalene Triangle.";
    }
  }

  public void compareTrinagles(Triangle MN){//method for comparing sides and peremeters between 2 triangle objects
    if(this==MN){
      System.out.println("These two triangle objects have the same address.");
    }
    else if((this.AR1==MN.AR1)&&(this.AR2==MN.AR2)&&(this.AR3==MN.AR3)){
      System.out.println("Addresses are different but the sides of the triangles are equal.");
    }
    else if(this.AD==MN.AD){
      System.out.println("Only the perimeter of both triangles is equal.");
    }
    else{
      System.out.println("Addresses, length of the sides and perimeter all are different.");
    }
  }
}

public class BracuBus{

  public int BN;//maximum passengers
  public int BM;//passenger count
  public String[]BA;//String for taking Passengers
  public BracuBus(String br){//contructor with one peremeter for taking bus route
    this.BR=br;
    this.BN=2;
    this.BM=0;
    this.BA=new String[2];
  }
  public BracuBus(String br, int bn){//constructor with two peremeter for taking bus route and max passenger number
    this.BR=br;
    this.BN=bn;
    this.BM=0;
    this.BA=new String[bn];
  }
  public void board(){//method for boarding on bus
    System.out.println("No Passenger");
  }
      
  public void board(BracuStudent MN){//method for boarding on bus for one passenger 
    if(MN.AK=false){
      System.out.println("You don't have a bus pass!");
    }
    else if(!MN.AN.equals(BR)){
      System.out.println("You got on the wrong bus!");
    }
    else if(BM>BN){
      System.out.println("Bus is full!");
    }
    else if(BM<BN){
      BA[BM]=MN.AR;
      BM++;
      System.out.println(MN.AR+" boarded the bus");
    }
    else{}
  }
  public void board(BracuStudent MN1,BracuStudent MN2){//method for boarding on bus for two passenger 
    board( MN1);
    board( MN2);
  }
  public void showDetails(){//method for showing bus information
    System.out.println("Bus Route:"+BR);
    System.out.println("Passengr Count"+BM+"(Max:"+BN+")");
    System.out.println("Passenger on Board");
    for(int LK=0;LK<BM;LK++){
      System.out.print(BA[LK]);
    }
  }
}
