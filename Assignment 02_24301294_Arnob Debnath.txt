public class Course{
  public String AR;//Initial Instance Variable1
  public String DP;//Initial Instance Variable2
  public int AD;//Initial Instance Variable3
  
  public void updateDetails(String DE,String DB, int DR){//creating method1
    AR=DE;//put method1 value in instance Variable1
    DP=DB;//put method2 value in instance Variable2
    AD=DR;//put method3 value in instance Variable3
  }
  public void displayCourse(){//creating method2
    System.out.println("Course Name:"+AR);//Showing output1
    System.out.println("Course Code:"+DP);//Showing output2
    System.out.println("Course Credit:"+AD);//Showing output3
  }
}

public class Assignment{
  public int tasks=0;//Initial Instance Variable1
  public String difficulty=null;//Initial Instance Variable2
  public boolean submission=false;//Initial Instance Variable3
  
  public void printDetails(){//Creating Method1
    System.out.println("Number of tasks:" +tasks);
    System.out.println("Difficulty level:" +difficulty); 
    System.out.println("Submission required:" +submission);
  }
  public String makeOptional(){//Creating Method2
    
    if(submission=false){//boolean condition checking
      return "Submission is already not required";
    }
    else{
      return "Assignment will not require submission";
    }
  }
}
