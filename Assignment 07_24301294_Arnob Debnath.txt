    super(am,ar,aw);
    this.AL=al;
  }
  public void calculateSalary(){//method for calculating salary
    if(AL.equals("Java")){
      PK=getBaseSalary()+700;
      setBaseSalary(PK);
     }
    else{
      PK=getBaseSalary();
      System.out.println("No additional salary added");
    }
  }
  public void displayInfo(){//method for showing information
    super.displayInfo();
    System.out.println("Language"+AL);
    System.out.println("Final Salary"+PK);
  }
}

public class Manager extends Employee{//Child class Manager inheriting from parent class Employee
  public double AB;//Instance varibale for Bonus
  public double AF=0;//Final salary
  public Manager(String AM,double AR,int AW,double ab){//constructor with 4 peremeter
    super(AM,AR,AW);
    this.AB=ab;
  }
  public void calculateSalary(){//method for calculating salary 
    double AJ=AB/100;//bonus in decimal
    if(getHoursWorked()>40){
      double AT=getBaseSalary()*AJ;//bonus percentage on salary
      AF=getBaseSalary()+AT;//final salary after adding bonus salary
      }
    else{
      System.out.println("No bonus added");
    }
    }
  public void requestIncrement(double an){//method for checking and adding increment 
    double PO=0;
    if(getHoursWorked()<80){
      System.out.println("Increment Denied");
    }
    else if(getHoursWorked()>80 && getHoursWorked()<100){
      System.out.println("$"+(an/2)+" Increment approved");
      PO=getBaseSalary()+an/2;
      setBaseSalary(PO);
      AF=AF+PO;
    }
    else{
     System.out.println("$"+an+" Increment approved"); 
      PO=getBaseSalary()+an;
      setBaseSalary(PO);
      AF=AF+PO;
    }
  }
  public void displayInfo(){//method for showing information
    super.displayInfo();
    System.out.println("Bonus:"+AB+"%");
    System.out.println("Final Salary"+ AF);
  }
}
