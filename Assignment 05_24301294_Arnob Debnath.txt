public class Playlist{
  public String DA;//Name of the playlist
  public Song DD;//1st song
  public int DB;//song count
  public int DO;//numbering of songs
  public Song DC;//current Song
  public Playlist(String da){//contructer with one peremeter that takes name of the playlist
    this.DA=da;
    this.DD=null;
    this.DB=0;
    this.DO=1;
    System.out.println(DA+"created");
    }
  public void addSong(Song MN){//method of taking songs
    if(DD==null){//if there is no 1st song
      this.DD=MN;
    }
    else{
      this.DC=this.DD;//1st song is the current song
      while(DC.NXT!=null){//while loop for linking songs 
        DC=DC.NXT;
      }
      DC.setasnext(MN);//set other songs to next songs
    }
    DB++;
    System.out.println(MN.AR+"added to"+DA);
  }
  public void info(){//method for showing detail information
    System.out.println(DA+"has the following songs:");
    if(DD==null){
      System.out.println("No songs in"+DA);
    }
    else{
      this.DC=this.DD;
      while(DC!=null){
        System.out.println("Song"+DO);
        System.out.println("Title"+DC.AR);
        System.out.println("Artist"+DC.AD);
        System.out.println("Length"+DC.AM+"minutes");
        DC=DC.NXT;
        DO++;
      }
    }
  }
}
